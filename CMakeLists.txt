cmake_minimum_required(VERSION 3.16)
project(Trajetoria VERSION 2.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(flight_cli native/main.cpp)
add_executable(flight_tests native/tests.cpp)
if(MSVC)
  target_compile_options(flight_cli PRIVATE /W4 /fp:precise)
  target_compile_options(flight_tests PRIVATE /W4 /fp:precise)
else()
  target_compile_options(flight_cli PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(flight_tests PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_test(NAME physics COMMAND flight_tests)
